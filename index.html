1. Core Operational KPI Card Metrics

Calculates overall shipments volume, On-Time Delivery Rate (OTDR), average Order Fill Rate (OFR), and Stockout frequencies.

SELECT
    COUNT(Order_ID) AS Total_Orders,
    ROUND(SUM(On_Time_Delivery) * 100.0 / COUNT(Order_ID), 2) AS On_Time_Delivery_Rate_OTDR,
    ROUND(AVG(Order_Fill_Rate_Pct), 2) AS Avg_Order_Fill_Rate,
    SUM(Stockout_Flag) AS Total_Stockouts,
    ROUND(SUM(Stockout_Flag) * 100.0 / COUNT(Order_ID), 2) AS Stockout_Frequency_Rate,
    ROUND(AVG(Supplier_Lead_Time_Days), 2) AS Avg_Supplier_Lead_Time_Days
FROM supply_chain_shipments;

2. Supplier Performance Scoring (CTE & Window Functions)

Scores and ranks suppliers based on OTDR, Order Fill Rate, and lead time variance, using DENSE_RANK.

WITH SupplierBaseMetrics AS (
    SELECT
        Supplier_ID, Supplier_Name,
        SUM(On_Time_Delivery) * 100.0 / COUNT(Order_ID) AS OTDR,
        AVG(Order_Fill_Rate_Pct) AS OFR,
        AVG(Supplier_Lead_Time_Days) AS Avg_Lead_Time
    FROM supply_chain_shipments
    GROUP BY Supplier_ID, Supplier_Name
),
SupplierScoring AS (
    SELECT *,
        (OTDR * 0.4) + (OFR * 0.4) + ((20 - Avg_Lead_Time) * 5.0 * 0.2) AS Performance_Score
    FROM SupplierBaseMetrics
)
SELECT *,
    DENSE_RANK() OVER (ORDER BY Performance_Score DESC) AS Performance_Rank
FROM SupplierScoring;

3. Disruption Hotspots & Financial Exposure Analysis

Identifies vulnerable lanes, counting disruption incidents and summing total shipping cost exposure.

SELECT
    Supplier_Region, Transportation_Mode, Disruption_Type,
    COUNT(Order_ID) AS Disrupted_Orders_Count,
    ROUND(AVG(Disruption_Impact_Days), 2) AS Avg_Disruption_Delay_Days,
    ROUND(SUM(Transportation_Cost_INR) / 100000.0, 2) AS Transport_Spent_Lakhs
FROM supply_chain_shipments
WHERE Disruption_Type <> 'None'
GROUP BY Supplier_Region, Transportation_Mode, Disruption_Type
ORDER BY Disrupted_Orders_Count DESC;
